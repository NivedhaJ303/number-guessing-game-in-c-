#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;

int main() {
    srand(time(0)); // See the random number generator
    int secret = rand() % 50 + 1; // Generates a random number between 1 and 50
    int guess, attempts = 0; // Initialize guess and attempts
    cout << "This is a Number Guessing Game!" << endl;
    cout << "You can select a number that ranges from 1 to 50." << endl;

    for (int i = 0; i < 5; i++) { // Allow up to 5 attempts
        cout << "Guess a number: ";
        cin >> guess; // Capture user input
        attempts++; // Increment the number of attempts

        if (guess == secret) { // Check if the guess is correct
            cout << "Congratulations, your guess is right!!" << endl;
            switch (attempts) {
                case 1: cout << "You have guessed the number in the first attempt, your score is 100." << endl;
 break;
                case 2: cout << "You have guessed the number in the second attempt, your score is 80." << endl;
 break;
                case 3: cout << "You have guessed the number in the third attempt, your score is 60." << endl; 
break;
                case 4: cout << "You have guessed the number in the fourth attempt, your score is 40." << endl; 
break;
                case 5: cout << "You have guessed the number in the fifth attempt, your score is 20." << endl; 
break;
            }
            cout << "Total number of trials: " << attempts << endl;
            return 0; // Exit the program
        } else if (guess > 0 && guess > secret) { // Provide feedback for high guesses
            cout << "Too high, pick a lower number." << endl;
        } else { // Provide feedback for low guesses
            cout << "Too low, pick a higher number." << endl;
        }
    }

    cout << "Total number of trials: " << attempts << endl;
    cout << "Only 5 attempts allowed, try next time!" << endl;

    return 0; // Exit the program
}
